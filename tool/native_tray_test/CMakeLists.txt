cmake_minimum_required(VERSION 3.14)
project(nearlink_tray_test LANGUAGES CXX)
get_filename_component(PROJECT_ROOT "../.." ABSOLUTE)
set(RUNNER "${PROJECT_ROOT}/windows/runner")
add_executable(nearlink_tray_test main.cpp "${RUNNER}/tray_inbox.cpp" "${RUNNER}/Runner.rc")
target_compile_features(nearlink_tray_test PRIVATE cxx_std_17)
target_compile_options(nearlink_tray_test PRIVATE /W4 /WX /utf-8 /EHsc)
target_compile_definitions(nearlink_tray_test PRIVATE NOMINMAX UNICODE _UNICODE)
target_include_directories(nearlink_tray_test PRIVATE "${RUNNER}" "${PROJECT_ROOT}/windows/flutter/ephemeral/cpp_client_wrapper/include")
target_link_libraries(nearlink_tray_test PRIVATE gdiplus ole32 comctl32 shell32 user32 gdi32)
